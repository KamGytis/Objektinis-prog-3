19void skaitymas_is_failo(const std::string& filename, Vector<Studentas>& studentai) {
20 std::ifstream file(filename);
21 if (!file.is_open())
22 throw std::runtime_error("Nepavyko atidaryti failo: " + filename);
23
24 studentai.clear();
25 std::string header, line;
26 std::getline(file, header);
27
28 int nr = 0;
29 while (std::getline(file, line)) {
30 ++nr;
31 std::stringstream ss(line);
32 std::string vardas, pavarde;
33 ss >> vardas >> pavarde;
34 Vector<int> paz;
35 int v;
36 while (ss >> v) paz.push_back(v);
37 if (paz.empty()) continue;
38 int egz = paz.back();
39 paz.pop_back();
40 studentai.emplace_back(vardas, pavarde, paz, egz);
41 if (nr % 1000000 == 0)
42 std::cout << " Nuskaityta: " << nr << " studentu...\n";
43 }
44 std::cout << " Is viso nuskaityta: " << studentai.size() << " studentu\n";
45}

80void rusiavimas(Vector<Studentas>& studentai, int budas) {
81 switch (budas) {
82 case 1: std::sort(studentai.begin(), studentai.end(),
83 [](const Studentas& a, const Studentas& b) { return a.getVardas() < b.getVardas(); }); break;
84 case 2: std::sort(studentai.begin(), studentai.end(),
85 [](const Studentas& a, const Studentas& b) { return a.getVardas() > b.getVardas(); }); break;
86 case 3: std::sort(studentai.begin(), studentai.end(),
87 [](const Studentas& a, const Studentas& b) { return a.getPavarde() < b.getPavarde(); }); break;
88 case 4: std::sort(studentai.begin(), studentai.end(),
89 [](const Studentas& a, const Studentas& b) { return a.getPavarde() > b.getPavarde(); }); break;
90 case 5: std::sort(studentai.begin(), studentai.end(),
91 [](const Studentas& a, const Studentas& b) { return a.getRez() < b.getRez(); }); break;
92 case 6: std::sort(studentai.begin(), studentai.end(),
93 [](const Studentas& a, const Studentas& b) { return a.getRez() > b.getRez(); }); break;
94 default: std::cerr << "Neteisingas rusiavimo budas.\n"; break;
95 }
96}

123double skirstymas_s1(const Vector<Studentas>& studentai,
124 Vector<Studentas>& kieti,
125 Vector<Studentas>& vargsai) {
126 auto t0 = std::chrono::high_resolution_clock::now();
127 kieti.clear(); vargsai.clear();
128 std::copy_if(studentai.begin(), studentai.end(), std::back_inserter(kieti),
129 [](const Studentas& s) { return s.getRez() >= 5.0; });
130 std::copy_if(studentai.begin(), studentai.end(), std::back_inserter(vargsai),
131 [](const Studentas& s) { return s.getRez() < 5.0; });
132 return std::chrono::duration<double>(
133 std::chrono::high_resolution_clock::now() - t0).count();
134}

158 return std::chrono::duration<double>(
