75 void reallocate(size_type new_cap) {
76 pointer new_data = AllocTraits::allocate(alloc_, new_cap);
77 size_type i = 0;
78 try {
79 for (; i < size_; ++i)
80 AllocTraits::construct(alloc_, new_data + i, std::move_if_noexcept(data_[i]));
81 }
82 catch (...) {

84 AllocTraits::destroy(alloc_, new_data + j);
85 AllocTraits::deallocate(alloc_, new_data, new_cap);

89 data_ = new_data;
90 size_ = i;

384 * @brief Įterpia count kopijų reikšmės prieš pos.
385 */
386 iterator insert(const_iterator pos, size_type count, const T& value) {
387 size_type idx = pos - cbegin();
388 if (count == 0) return begin() + idx;
389 if (size_ + count > capacity_) {
390 size_type new_cap = std::max(size_ + count, capacity_ * 2);
391 reallocate(new_cap);
392 }
393
394 for (size_type i = size_; i > idx; --i) {
395 size_type old_idx = i - 1;
396 size_type new_idx = old_idx + count;
397 if (new_idx >= size_) {
398 AllocTraits::construct(alloc_, data_ + new_idx, std::move(data_[old_idx]));
399 }
400 else {
401 data_[new_idx] = std::move(data_[old_idx]);
402 }
403 }
404
405 size_type affected_constructed = std::min(size_, idx + count);
406 for (size_type i = idx; i < affected_constructed; ++i) {
407 AllocTraits::destroy(alloc_, data_ + i);
408 }
409
410 for (size_type i = 0; i < count; ++i) {
411 AllocTraits::construct(alloc_, data_ + idx + i, value);

458 /**
459 * @brief Sukonstruoja elementą vietoje prieš pos.
460 */
461 template<typename... Args>
462 iterator emplace(const_iterator pos, Args&&... args) {
463 size_type idx = pos - cbegin();
464 if (size_ == capacity_)
465 reallocate(capacity_ == 0 ? 1 : capacity_ * 2);
466
467 for (size_type i = size_; i > idx; --i) {
468 size_type old_idx = i - 1;
469 size_type new_idx = old_idx + 1;
470 if (new_idx >= size_) {
471 AllocTraits::construct(alloc_, data_ + new_idx, std::move(data_[old_idx]));
472 }
473 else {
474 data_[new_idx] = std::move(data_[old_idx]);
475 }
476 }
477
478 if (idx < size_) {
479 AllocTraits::destroy(alloc_, data_ + idx);
480 }
481
482 AllocTraits::construct(alloc_, data_ + idx, std::forward<Args>(args)...);
483 ++size_;
484 return begin() + idx;
485 }
