259static void testuoti_su_stl() {
260 sekcija("Naudojimas su STL konteineriais ir algoritmais");
261
262 std::vector<Studentas> studentai = {
263 Studentas("C", "C", {4,4}, 4),
264 Studentas("A", "A", {9,9}, 9),
265 Studentas("B", "B", {6,6}, 6)
266 };
267 for (auto& s : studentai) s.skaiciuotiRez(1);
268
269 // std::sort su operator<
270 std::sort(studentai.begin(), studentai.end());
271 tikrinti(studentai[0].getVardas() == "C" &&
272 studentai[1].getVardas() == "B" &&
273 studentai[2].getVardas() == "A",
274 "std::sort su operator<: rusiuoja pagal rez_");
275
276 // std::find_if su islaike
277 auto pirmas_islaikes = std::find_if(studentai.begin(), studentai.end(),
278 [](const Studentas& s) { return s.islaike(); });
279 tikrinti(pirmas_islaikes != studentai.end() &&
280 pirmas_islaikes->getVardas() == "B",
281 "std::find_if su islaike(): randa pirma islaikiusi");
282 // std::partition su operator>=
283 std::vector<Studentas> v = studentai;
284 auto pivot = std::partition(v.begin(), v.end(),
285 [](const Studentas& s) { return s.islaike(); });
286 int islaike = (int)std::distance(v.begin(), pivot);
287 int neislaike = (int)std::distance(pivot, v.end());
288 tikrinti(islaike == 2 && neislaike == 1,
289 "std::partition + islaike(): 2 islaike, 1 neislaike");
290}
